#include <SPI.h>
#include <DW1000.h>

// --- KONFIGURASI ANCHOR ---
const uint8_t ANCHOR_ID = 1; // Ubah jadi 2 untuk Anchor kedua
// --------------------------

const uint8_t PIN_RST = 4;
const uint8_t PIN_IRQ = 17;
const uint8_t PIN_SS  = 16;

volatile boolean received = false;
volatile boolean error = false;

// Struktur harus sama persis dengan Tag
// Hemat 4 byte dengan membuang txTime
struct BlinkPacket {
  uint8_t frameType; // 1 byte (0xA0)
  uint8_t tagId;     // 1 byte
  uint32_t seq;      // 4 byte
};
// Total payload: 6 Bytes. Sangat cepat!

BlinkPacket rxPacket;

void handleReceived() {
  received = true;
}

void handleError() {
  error = true;
}

void setup() {
  Serial.begin(115200); // Speed tinggi untuk minimalisir latency
  delay(1000);

  SPI.begin(18, 19, 23);
  pinMode(PIN_RST, OUTPUT);
  digitalWrite(PIN_RST, LOW); delay(50);
  digitalWrite(PIN_RST, HIGH); delay(500);

  DW1000.begin(PIN_IRQ, PIN_RST);
  DW1000.select(PIN_SS);
  
  DW1000.newConfiguration();
  DW1000.setDefaults();
  DW1000.setDeviceAddress(ANCHOR_ID);
  DW1000.setNetworkId(10);
  DW1000.enableMode(DW1000.MODE_LONGDATA_RANGE_LOWPOWER);
  DW1000.commitConfiguration();
  
  DW1000.attachReceivedHandler(handleReceived);
  DW1000.attachReceiveFailedHandler(handleError);
  DW1000.attachErrorHandler(handleError);
  
  // Mulai Receive Permanen
  DW1000.newReceive();
  DW1000.setDefaults();
  DW1000.receivePermanently(true);
  DW1000.startReceive();
  
  Serial.println("### ANCHOR TDoA LISTENER READY ###");
  Serial.println("FORMAT: ANCHOR_ID, TAG_ID, SEQ, RX_TIMESTAMP_40BIT_HEX");
}

void loop() {
if (received) {
    received = false;
    
    // Validasi panjang data harus 6 byte
    if (DW1000.getDataLength() == sizeof(BlinkPacket)) {
      DW1000.getData((byte*)&rxPacket, sizeof(BlinkPacket));
      
      if (rxPacket.frameType == 0xA0) {
        
        // Ambil waktu kedatangan (INI YANG PENTING BUAT TDoA)
        DW1000Time rxTime;
        DW1000.getReceiveTimestamp(rxTime);
        
        // Print ke Python: ANCHOR, TAG, SEQ, RX_TIME
        Serial.print(ANCHOR_ID);
        Serial.print(",");
        Serial.print(rxPacket.tagId);
        Serial.print(",");
        Serial.print(rxPacket.seq);
        Serial.print(",");
        
        // Cetak RX Timestamp (5 byte Hex presisi tinggi)
        byte ts[5];
        rxTime.getTimestamp(ts);
        for(int i=4; i>=0; i--) {
          if(ts[i] < 16) Serial.print("0");
          Serial.print(ts[i], HEX);
        }
        Serial.println();
      }
    }
  }

  if (error) {
    error = false;
    // Serial.println("Err"); // Optional: Debugging
    // DW1000 menangani restart receive otomatis via receivePermanently
  }
}
